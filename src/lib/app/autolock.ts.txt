import { session } from './session';
import { get } from 'svelte/store';

import { lockIsSuspended,uiGuard } from './uiGuard';
// ...
function maybeLockOnBlurOrHide() {
  if (lockIsSuspended()) return;           // <-- ignore while WebAuthn is running
  // lock(); // your existing lock call
}


let timeout: number | null = null;

export function startAutoLock(minutes = 10) {
  const schedule = () => {
    if (timeout) window.clearTimeout(timeout);
    const s = get(session);
    if (s.key) timeout = window.setTimeout(() => lockNow(), minutes * 60_000);
  };
  const reset = () => schedule();

  schedule();
  window.addEventListener('mousemove', reset);
  window.addEventListener('keydown', reset);
  console.log('[mvault] Auto-lock on blur is suspended due to ongoing operation:', get(uiGuard).reason);
  if (!lockIsSuspended())            // <-- ignore while WebAuthn is running
    window.addEventListener('blur', lockNow);
  else
    window.removeEventListener('blur', lockNow);



  return () => {
    if (timeout) window.clearTimeout(timeout);
    window.removeEventListener('mousemove', reset);
    window.removeEventListener('keydown', reset);
    window.removeEventListener('blur', lockNow);
  };
}

function lockNow() {
  import('./session').then(({ lock }) => lock());
}
